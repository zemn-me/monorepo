cmake_minimum_required(VERSION 3.23)
project(OpenTSWeb LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -msimd128 -fno-fast-math -ffp-contract=off -Wno-deprecated-declarations")
add_subdirectory(thirdparty)
file(GLOB_RECURSE ENGINE CONFIGURE_DEPENDS code/*.cpp)
list(FILTER ENGINE EXCLUDE REGEX "/language/dllmain.cpp$")
list(FILTER ENGINE EXCLUDE REGEX "/netsocket_win32.cpp$")
add_executable(opents ${ENGINE} web/platform.cpp)
target_include_directories(opents PRIVATE code web generated code/vqalib)
target_compile_options(opents PRIVATE -fms-extensions -include ${CMAKE_SOURCE_DIR}/web/platform.h)
target_compile_definitions(opents PRIVATE NOMINMAX)
target_include_directories(opents PRIVATE thirdparty/bgfx.cmake/bgfx/include thirdparty/bgfx.cmake/bx/include)
set_source_files_properties(code/bgfxbackend.cpp PROPERTIES INCLUDE_DIRECTORIES "${CMAKE_SOURCE_DIR}/thirdparty/bgfx.cmake/bgfx/examples/common/imgui")
set_source_files_properties(code/ui/rml/rmlrender.cpp PROPERTIES INCLUDE_DIRECTORIES "${CMAKE_SOURCE_DIR}/thirdparty/bgfx.cmake/bgfx/examples/common/debugdraw")
set_source_files_properties(code/ui/rml/rmltexture.cpp PROPERTIES INCLUDE_DIRECTORIES "${CMAKE_SOURCE_DIR}/thirdparty/bgfx.cmake/bimg/3rdparty/stb")
set_source_files_properties(code/writepng.cpp PROPERTIES INCLUDE_DIRECTORIES "${CMAKE_SOURCE_DIR}/thirdparty/bgfx.cmake/bimg/3rdparty/tinyexr/deps")
target_compile_definitions(opents PRIVATE BX_CONFIG_DEBUG=0)
target_link_libraries(opents PRIVATE bgfx bx bimg miniaudio lzo RmlUi::Core freetype imgui SDL3::SDL3-static)
target_link_options(opents PRIVATE -sASYNCIFY=1 -sALLOW_MEMORY_GROWTH=1 -sMAXIMUM_MEMORY=2147483648 -sINITIAL_MEMORY=268435456 -sSTACK_SIZE=8388608 -sMODULARIZE=1 -sEXPORT_ES6=1 -sEXPORTED_RUNTIME_METHODS=FS,IDBFS,callMain,requestFullscreen -lidbfs.js -sINVOKE_RUN=0 -sFORCE_FILESYSTEM=1 -sEXIT_RUNTIME=1 -sENVIRONMENT=web -sMIN_WEBGL_VERSION=2 -sMAX_WEBGL_VERSION=2 -sASSERTIONS=1 --preload-file ${CMAKE_SOURCE_DIR}/ui@/game/ui)
set_target_properties(opents PROPERTIES SUFFIX ".js")
